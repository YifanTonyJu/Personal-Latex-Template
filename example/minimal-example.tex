\section{Mathematical Proofs}

\subsection{Theorems (已有的著名定理)}

\begin{theorem}
    For any positive real numbers $a$ and $b$, the arithmetic mean is at least the geometric mean.
\end{theorem}

\begin{proof}
    By expanding $(a-b)^2 \geq 0$, we get $a^2 + b^2 \geq 2ab$.
    Dividing by $2ab$ yields $\frac{a+b}{2} \geq \sqrt{ab}$.
\end{proof}

\subsection{Propositions (个人结论)}

\begin{proposition}
    The square of any odd integer is congruent to 1 modulo 8.
\end{proposition}

\begin{proof}
    Let $n = 2k+1$ be an odd integer. Then $n^2 = (2k+1)^2 = 4k^2 + 4k + 1 = 4k(k+1) + 1$.

    Since $k(k+1)$ is always even, $4k(k+1) \equiv 0 \pmod{8}$, so $n^2 \equiv 1 \pmod{8}$.
\end{proof}

\subsection{Lemmas (辅助引理)}

\begin{lemma}
    For any integer $n$, either $n$ or $n+1$ is even.
\end{lemma}

\begin{prooflemma}
    By the division algorithm, every integer is either even or odd.
\end{prooflemma}

\subsection{Claims (临时小台阶)}

\begin{claim}
    The sum of two even numbers is even.
\end{claim}

\begin{proofclaim}
    Let $2m$ and $2n$ be two even numbers. Their sum is $2m + 2n = 2(m+n)$, which is even.
\end{proofclaim}
