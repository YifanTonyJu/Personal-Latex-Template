Input: Array A, left index p, right index r
Output: Sorted array A[p...r]

if p < r then
    q := PARTITION(A, p, r)
    QUICKSORT(A, p, q - 1)
    QUICKSORT(A, q + 1, r)
end if

Algorithm: PARTITION(A, p, r)
Input: Array A, left index p, right index r
Output: Partition index

x := A[r]
i := p - 1
for j := p to r - 1 do
    if A[j] <= x then
        i := i + 1
        swap A[i] and A[j]
    end if
end for
swap A[i + 1] and A[r]
return i + 1
